## Test Execution Script

Save as `test-r8s-features.sh`:

```bash
#!/bin/bash
# R8S Feature Validation Script
# Usage: ./test-r8s-features.sh /path/to/bundle1 /path/to/bundle2 ...

set -e

R8S_BINARY="${R8S_BINARY:-r8s}"
TEST_RESULTS="test-results-v1.0.1-$(date +%Y%m%d-%H%M%S)"
mkdir -p "$TEST_RESULTS"

echo "═══════════════════════════════════════════════════════════"
echo "  R8S v1.0.1 Feature Validation"
echo "  Started: $(date)"
echo "  Binary: $($R8S_BINARY version 2>/dev/null || echo 'NOT FOUND')"
echo "═══════════════════════════════════════════════════════════"
echo

# Test bundle paths from args or default
BUNDLES=(${@:-$(ls -d ~/bundles/*/ 2>/dev/null || echo '')})

if [ ${#BUNDLES[@]} -eq 0 ]; then
    echo "ERROR: No test bundles provided"
    echo "Usage: $0 /path/to/bundle1 /path/to/bundle2 ..."
    exit 1
fi

PASS=0
FAIL=0

test_feature() {
    local test_name=$1
    local command=$2
    local check=$3
    local bundle=$4
    
    echo "─────────────────────────────────────────────────────────"
    echo "TEST: $test_name"
    echo "COMMAND: $command"
    echo
    
    local output="$TEST_RESULTS/${test_name// /_}.txt"
    local start_time=$(date +%s)
    
    # Run command
    if eval "$command" > "$output" 2>&1; then
        local exit_code=0
    else
        local exit_code=$?
    fi
    
    local end_time=$(date +%s)
    local duration=$((end_time - start_time))
    
    # Check result
    if [ $exit_code -eq 0 ] && eval "$check"; then
        echo "✅ PASS (${duration}s)"
        ((PASS++))
        return 0
    else
        echo "❌ FAIL (exit: $exit_code, ${duration}s)"
        echo "   Output preview:"
        head -5 "$output" | sed 's/^/   /'
        ((FAIL++))
        return 1
    fi
}

for bundle in "${BUNDLES[@]}"; do
    echo
    echo "═══════════════════════════════════════════════════════════"
    echo "  Testing Bundle: $(basename "$bundle")"
    echo "  Path: $bundle"
    echo "═══════════════════════════════════════════════════════════"
    echo
    
    # TEST 1: Version
    test_feature "version" \
        "$R8S_BINARY version" \
        "true" \
        "$bundle"
    
    # TEST 2: Help
    test_feature "help" \
        "$R8S_BINARY --help" \
        "true" \
        "$bundle"
    
    # TEST 3: Analyze (critical - must find issues if present)
    test_feature "analyze" \
        "$R8S_BINARY analyze $bundle" \
        "grep -q 'Bundle:' $TEST_RESULTS/analyze.txt 2>/dev/null" \
        "$bundle"
    
    # TEST 4: Analyze JSON output
    test_feature "analyze-json" \
        "$R8S_BINARY analyze $bundle --format=json" \
        "grep -q 'bundle_type' $TEST_RESULTS/analyze-json.txt 2>/dev/null" \
        "$bundle"
    
    # TEST 5: Validate bundle
    test_feature "validate" \
        "$R8S_BINARY validate $bundle" \
        "true" \
        "$bundle"
    
    # TEST 6: Get pods
    test_feature "get-pods" \
        "$R8S_BINARY get pods $bundle" \
        "true" \
        "$bundle"
    
    # TEST 7: Get nodes
    test_feature "get-nodes" \
        "$R8S_BINARY get nodes $bundle" \
        "true" \
        "$bundle"
    
    # TEST 8: Export SARIF (if issues found)
    test_feature "export-sarif" \
        "$R8S_BINARY export $bundle --format=sarif --output=$TEST_RESULTS/$(basename $bundle).sarif.json" \
        "[ -f $TEST_RESULTS/$(basename $bundle).sarif.json ]" \
        "$bundle"
    
    # TEST 9: Export Markdown
    test_feature "export-md" \
        "$R8S_BINARY export $bundle --format=markdown --output=$TEST_RESULTS/$(basename $bundle).md" \
        "[ -f $TEST_RESULTS/$(basename $bundle).md ]" \
        "$bundle"
    
    # TEST 10: Test cluster command
    test_feature "test-cluster" \
        "$R8S_BINARY test-cluster $bundle" \
        "true" \
        "$bundle"
    
    # TEST 11: Ask (if implemented)
    # Note: This may fail if 'ask' is not properly implemented
    test_feature "ask" \
        "$R8S_BINARY ask $bundle 'what issues are present?'" \
        "grep -q -i 'issue\|error\|warn\|crash\|oom\|image' $TEST_RESULTS/ask.txt 2>/dev/null || true" \
        "$bundle" || true  # Don't fail on ask for now
    
    # TEST 12: Get logs (first pod)
    FIRST_POD=$($R8S_BINARY get pods $bundle --format=json 2>/dev/null | jq -r '.items[0].metadata.name' 2>/dev/null || echo '')
    if [ -n "$FIRST_POD" ]; then
        test_feature "get-logs-$FIRST_POD" \
            "$R8S_BINARY logs $bundle $FIRST_POD" \
            "true" \
            "$bundle" || true
    fi
    
    # kubectl plugin tests (if available)
    if command -v kubectl &> /dev/null && kubectl plugin list 2>/dev/null | grep -q r8s; then
        export R8S_BUNDLE="$bundle"
        
        test_feature "kubectl-plugin-analyze" \
            "kubectl r8s analyze" \
            "true" \
            "$bundle" || true
    fi
done

echo
echo "═══════════════════════════════════════════════════════════"
echo "  TEST SUMMARY"
echo "═══════════════════════════════════════════════════════════"
echo "  Total: $((PASS + FAIL))"
echo "  Pass:  $PASS"
echo "  Fail:  $FAIL"
echo
echo "  Results: $TEST_RESULTS/"
echo
echo "  Duration: $(date +%Y%m%d-%H%M%S)"
echo "═══════════════════════════════════════════════════════════"

if [ $FAIL -gt 0 ]; then
    echo
echo "FAILED tests:"
    grep -l "❌ FAIL" $TEST_RESULTS/*.txt 2>/dev/null | sed 's/^/  - /' || true
    echo
    echo "View details:"
    echo "  cat $TEST_RESULTS/*.txt"
    echo
    exit 1
fi

echo "All tests PASSED ✅"
exit 0
```

## Expected Outputs Checklist

For each test, verify:

### analyze
- [ ] Shows bundle type (RKE2/K3s)  
- [ ] Shows health percentage  
- [ ] **CRITICAL:** Identifies known issues in the bundle  
- [ ] Lists severity counts (Critical/Warning/Info)

### ask
- [ ] Returns a non-empty response  
- [ ] References actual issues from the bundle  
- [ ] Provides actionable information

### get pods/nodes  
- [ ] Lists resources in table format  
- [ ] JSON output parsable

### export
- [ ] SARIF: Valid JSON with runs/results structure  
- [ ] Markdown: Human-readable with headers and lists  

### validate
- [ ] Reports bundle completeness percentage  
- [ ] Lists missing files if incomplete
